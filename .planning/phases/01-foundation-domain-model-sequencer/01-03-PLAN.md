```swift
public struct Camelot: Equatable, Hashable, Sendable {
    public let hour: Int      // 1...12
    public let isMinor: Bool  // A = minor, B = major
    public init?(_ code: String) {
        guard let last = code.last, last == "A" || last == "B",
              let n = Int(code.dropLast()), (1...12).contains(n) else { return nil }
        self.hour = n; self.isMinor = (last == "A")
    }
    public var code: String { "\(hour)\(isMinor ? "A" : "B")" }
    public func distance(to other: Camelot) -> Int {
        let d = abs(hour - other.hour)
        return min(d, 12 - d) + (isMinor == other.isMinor ? 0 : 1)
    }
}
```

Verified passing: same key 0, ±1 hour 1, relative major/minor 1, wrap 12→1 == 1, and `nil` on
both `"nonsense"` and `"13A"`.

`parse` order of attempts, which matters:
1. trim whitespace; empty or nil → nil
2. try `Camelot(code:)` — handles "4A", "12B"
3. musical: strip a trailing `"m"`/`"min"`/`"minor"` (case-insensitive) to decide minor, take the
   remainder as the pitch, normalise the accidental (`"eb"` → `"Eb"`), look it up in the
   matching table
4. anything left → nil

```swift
public enum BPMRules {
    public static let defaultTolerance: Double = 4.0
    public static let toleranceRange: ClosedRange<Double> = 3.0...6.0

    /// D6: jungle/DnB-only app — an 87 BPM read is a 174 BPM track. Applied once, not iteratively.
    public static func fold(_ bpm: Double) -> Double { bpm < 100 ? bpm * 2 : bpm }

    /// D2: the sequence's only hard gate. Non-decreasing AND within tolerance.
    public static func canFollow(_ a: Double, _ b: Double, tolerance: Double) -> Bool {
        b >= a && (b - a) <= tolerance + 1e-9
    }

    public static func clampTolerance(_ t: Double) -> Double {
        min(max(t, toleranceRange.lowerBound), toleranceRange.upperBound)
    }
}
```

The `1e-9` is deliberate: without it, a tolerance the user drags to 4.0 on a slider can reject a
gap that is 4.0 in decimal but 4.0000000000000004 in binary. The `174.01` test above still fails
correctly because the excess is seven orders of magnitude larger than the epsilon.

Add a doc comment on `canFollow` naming the known ceiling: monotonic BPM is an MVP
simplification (real sets arc warm-up → peak → cool-down). It is `SEQ2-01`, not a truth.
