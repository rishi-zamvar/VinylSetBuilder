  <action>
```swift
public struct BeamDivergence: Sendable {
    public let narrow: SequenceResult
    public let wide: SequenceResult
    /// True when the wider beam found a longer sequence, or an equally long one with a
    /// strictly lower total Camelot distance. A different-but-equivalent path is NOT divergence.
    public var diverged: Bool {
        wide.sequence.count > narrow.sequence.count
            || (wide.sequence.count == narrow.sequence.count
                && wide.totalCamelotDistance < narrow.totalCamelotDistance)
    }
    public var summary: String { ... }   // "width 50: 38 vinyls / 21 hops — width 150: 38 / 21"
}

public enum BeamDivergenceCheck {
    public static func compare(candidates: [Track], bpmTolerance: Double,
                               narrowWidth: Int = 50, wideWidth: Int = 150) -> BeamDivergence
}
```

Note the asymmetry in `diverged`: the wide beam is a superset search, so it can only ever match
or beat the narrow one. Comparing paths element-by-element would flag harmless ties between two
equally good orderings as failures.

`summary` exists so the failure message is useful. Have the tests print it unconditionally on
failure via `#expect(..., "\(divergence.summary)")`.
