import Foundation
import Testing

private let bannedTokens = [
    "AVFoundation", "Vision", "UIKit", "AppKit", "SwiftUI", "SwiftData",
    "URLSession", "URLRequest", "URLProtocol",
]

private var sourcesDirectory: URL {
    URL(fileURLWithPath: #filePath)      // .../Tests/VinylSetBuilderCoreTests/CorePurityTests.swift
        .deletingLastPathComponent()      // .../Tests/VinylSetBuilderCoreTests
        .deletingLastPathComponent()      // .../Tests
        .deletingLastPathComponent()      // .../VinylSetBuilderCore
        .appendingPathComponent("Sources")
}

@Test("SETUP-02: Core links no camera and no network framework")
func test_givenCoreSources_whenScanned_thenNoCameraOrNetworkSymbolAppears() throws {
    let fm = FileManager.default
    let enumerator = try #require(fm.enumerator(at: sourcesDirectory,
                                                includingPropertiesForKeys: nil))
    var scanned = 0
    var violations: [String] = []
    for case let url as URL in enumerator where url.pathExtension == "swift" {
        scanned += 1
        let text = try String(contentsOf: url, encoding: .utf8)
        for token in bannedTokens where text.contains(token) {
            violations.append("\(url.lastPathComponent): \(token)")
        }
    }
    #expect(scanned > 0, "purity guard scanned no files — the path walk is wrong")
    #expect(violations.isEmpty, "Core purity violated: \(violations.joined(separator: ", "))")
}
```

Note the `scanned > 0` assertion — without it the test passes vacuously if the path walk
breaks, which is exactly how this class of guard silently rots.

public enum Side: String, Sendable, Codable, Hashable, CaseIterable { case a, b }

public enum ResolutionState: String, Sendable, Codable, Hashable {
    case pending, resolved, failed, manual
}

public struct Track: Identifiable, Sendable, Hashable, Codable {
    public let id: UUID
    public let vinylID: UUID
    public var side: Side
    public var position: Int            // D9: explicit ordering, never relationship order
    public var rawOCRText: String
    public var artist: String?
    public var title: String?
    public var bpm: Double?
    public var camelot: String?         // "8A", "11B" — parsed by Camelot in plan 01-03
    public var matchConfidence: Double
    public var resolutionState: ResolutionState
    public var failureReason: String?
    public init(id: UUID = UUID(), vinylID: UUID, side: Side = .a, position: Int = 0,
                rawOCRText: String = "", artist: String? = nil, title: String? = nil,
                bpm: Double? = nil, camelot: String? = nil, matchConfidence: Double = 0,
                resolutionState: ResolutionState = .pending, failureReason: String? = nil)
    /// Display name for a UI row. Falls back to raw OCR text when unresolved.
    public var displayName: String { ... }   // "\(artist) — \(title)" or rawOCRText or "Unknown track"
}

public struct VinylEntry: Identifiable, Sendable, Hashable, Codable {
    public let id: UUID
    public var capturedAt: Date
    public var photoRefs: [String]       // D9: scalar array, order preserved as inserted
    public var sidesScanned: Set<Side>
    public var candidates: [Track]
    public var position: Int             // D9: order within its crate
    public var orderedCandidates: [Track] { candidates.sorted { $0.position < $1.position } }
    public var label: String { ... }     // best-effort display name for the vinyl
}

public struct Crate: Identifiable, Sendable, Hashable, Codable {
    public let id: UUID
    public var name: String
    public var createdAt: Date
    public var vinyls: [VinylEntry]
    public var bpmTolerance: Double                 // default 4.0, slider range 3...6
    public var manualOrder: [Track.ID]?             // non-nil once the user drags (CRATE-04)
    public var selectedTrackByVinyl: [UUID: UUID]   // CRATE-07: selection is per-crate, not per-vinyl
    public var recordCount: Int { vinyls.count }
    public var allCandidates: [Track] { vinyls.flatMap(\.orderedCandidates) }
}

public struct TextObservation: Sendable, Hashable {
    public let text: String
    public let boundingBox: CGRect      // normalized [0,1], Vision's convention
    public let confidence: Float
    public let baselineAngle: Double    // radians; Phase 3 clusters on this
}

public struct NotIncluded: Sendable, Hashable, Codable, Identifiable {
    public var id: UUID { vinylID }
    public let vinylID: UUID
    public let reason: String           // D7: plain language, shown verbatim in the UI
}

public struct SequenceResult: Sendable, Hashable {
    public let sequence: [Track]              // ordered, one per vinyl, non-decreasing BPM
    public let notIncluded: [NotIncluded]     // D7: ONE bucket for unresolved + excluded
    public let totalCamelotDistance: Int
    public static let empty = SequenceResult(sequence: [], notIncluded: [], totalCamelotDistance: 0)
}
```

Do NOT add SwiftData annotations. Phase 6 owns persistence; these stay plain structs.

Define the seams exactly as `.planning/research/ARCHITECTURE.md` specifies. These four are the
entire public surface later phases replace implementations behind.

```swift
public protocol Sequencing: Sendable {
    /// Pure function. Fully re-derives the sequence every call — no memory of prior runs.
    /// That is what makes SEQ-08 (rejoin on add) fall out for free.
    func sequence(candidates: [Track], bpmTolerance: Double) -> SequenceResult
}

public protocol LabelSegmenting: Sendable {
    /// Pure geometry. No I/O. Phase 3 supplies the real implementation.
    func segment(_ observations: [TextObservation], vinylID: UUID, side: Side) -> [Track]
}

public protocol MetadataResolving: Sendable {
    /// Takes a .pending Track, returns it resolved or with resolutionState == .failed.
    func resolve(_ track: Track) async -> Track
}
```

`LabelTextRecognizing` is deliberately NOT declared here — it takes a `CGImage` and belongs to
the app target's Vision layer in Phase 2. Declaring it in Core would put a Vision-shaped hole
in a package that is not allowed to import Vision.

Two stubs so the Phase-1 UI has something real to bind to:

- `FixtureMetadataResolver: MetadataResolving` — sleeps 1.5 s (`try? await Task.sleep(for: .seconds(1.5))`),
  then fills artist/title/bpm/camelot and sets `.resolved`. Make the returned values vary by
  `track.position` rather than being one constant, so a UI list of stubbed tracks does not look
  like a rendering bug.
- `NaiveBPMSortSequencer: Sequencing` — filters nil BPM, sorts ascending, returns them all with
  an empty `notIncluded`. This is the placeholder the crate UI uses until plan 01-04's
  `BeamSearchSequencer` exists; it exists so plan 01-06 does not block on plan 01-04.

No tests required for the stubs beyond the fact that they compile as conformances — that is
the only property they need to have.
