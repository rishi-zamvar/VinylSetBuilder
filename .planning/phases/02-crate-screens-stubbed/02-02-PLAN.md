The failure mode this task exists to avoid: XCUITest screenshots taken as `XCTAttachment` live
inside `build/ui.xcresult` and getting them out depends on `xcresulttool`'s export subcommand,
whose spelling has churned across Xcode versions. So write the PNG **twice** — once to the test
runner's own Documents directory (deterministic, copied out with `simctl get_app_container`) and
once as an attachment (free, and it makes a failing run debuggable).

`UITests/ScreenshotRecorder.swift`:

```swift
import XCTest

/// Every screenshot this phase ships goes through here, so the settle delay is applied in
/// exactly one place and cannot be forgotten on one state out of fifteen.
///
/// V2 was closed on 2026-08-30: the washed-out UI was the launch fade, not styling. The rule
/// it left behind is mandatory — a screenshot taken mid-transition is evidence of nothing.
enum ScreenshotRecorder {
    /// `.planning/VISUAL-DEFECTS.md` closed V2 on 2026-08-30 and left a MANDATORY rule behind:
    /// every screenshot waits at least 3 seconds. Do not lower these. The original V2 report
    /// was a launch-fade artefact that nearly caused correct, well-contrasted code to be
    /// "fixed" into a real regression.
    static let defaultSettle: TimeInterval = 3.0
    static let launchSettle: TimeInterval = 4.0

    static var outputDirectory: URL {
        FileManager.default.urls(for: .documentDirectory, in: .userDomainMask)[0]
    }

    @discardableResult
    static func capture(_ name: String,
                        settle: TimeInterval = defaultSettle,
                        in testCase: XCTestCase) -> URL? {
        Thread.sleep(forTimeInterval: settle)
        let shot = XCUIScreen.main.screenshot()

        let attachment = XCTAttachment(screenshot: shot)
        attachment.name = name
        attachment.lifetime = .keepAlways
        testCase.add(attachment)

        let url = outputDirectory.appendingPathComponent("\(name).png")
        do { try shot.pngRepresentation.write(to: url); return url }
        catch { XCTFail("could not write \(name).png: \(error)"); return nil }
    }
}
```

`Thread.sleep` in the test process is correct here: the app under test is a *separate* process
and keeps running. This is the settle delay the ship criterion demands — it is not a workaround.

Add one screenshot call to `SmokeUITests`, proving the loop end to end:
`ScreenshotRecorder.capture("02-smoke-crate-list", settle: ScreenshotRecorder.launchSettle, in: self)`

`Scripts/collect-screenshots.sh`:

```bash
#!/usr/bin/env bash
# Copies the PNGs the XCUITest wrote inside the simulator out onto the host.
set -euo pipefail
cd "$(dirname "$0")/.."
UDID="4B4E1C1F-BED6-4812-9238-C151337265DA"
RUNNER="com.rishizamvar.VinylSetBuilderUITests.xctrunner"

CONTAINER="$(xcrun simctl get_app_container "$UDID" "$RUNNER" data 2>/dev/null || true)"
if [ -z "$CONTAINER" ]; then
  # Fallback: the runner bundle id is derived by Xcode and can differ. Find it.
  RUNNER="$(xcrun simctl listapps "$UDID" | grep -o '"[^"]*xctrunner"' | tr -d '"' | head -1)"
  CONTAINER="$(xcrun simctl get_app_container "$UDID" "$RUNNER" data)"
fi

mkdir -p Artifacts/screenshots
COUNT=$(ls "$CONTAINER"/Documents/*.png 2>/dev/null | wc -l | tr -d ' ')
[ "$COUNT" -gt 0 ] || { echo "no screenshots in $CONTAINER/Documents"; exit 1; }
cp "$CONTAINER"/Documents/*.png Artifacts/screenshots/
echo "collected $COUNT screenshot(s) into Artifacts/screenshots/"
```
