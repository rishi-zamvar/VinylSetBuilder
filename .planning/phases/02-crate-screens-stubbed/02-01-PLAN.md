```swift
import Foundation

/// CRATE-03/04/07 as a pure value type: the running order, the per-crate pin, and the manual
/// override. Holds no reference to any view layer (D8) and performs no work in `init` —
/// a caller sequences explicitly, because a caller is the one who knows what it can afford.
public struct CrateDetailModel: Sendable, Equatable {
    public private(set) var crate: Crate
    public private(set) var result: SequenceResult
    public private(set) var rows: CrateRows

    public init(crate: Crate) {
        self.crate = crate
        self.result = .empty
        self.rows = .empty
    }

    public var crateID: UUID { crate.id }
    public var crateName: String { crate.name }
    public var recordCount: Int { crate.recordCount }

    /// CRATE-07 (D2-2). A per-crate selection PINS which candidate the sequencer may use for
    /// that vinyl — that is what makes the same vinyl render differently in two crates. A pin
    /// naming a track the vinyl does not actually carry is ignored rather than trusted, so a
    /// stale selection can never erase a vinyl from the set.
    public var sequencerInput: [Track] {
        crate.vinyls.flatMap { vinyl -> [Track] in
            let candidates = vinyl.orderedCandidates
            guard let pinned = crate.selectedTrackByVinyl[vinyl.id],
                  let track = candidates.first(where: { $0.id == pinned })
            else { return candidates }
            return [track]
        }
    }

    /// Apply a result computed elsewhere. Rebuilds `rows` and reconciles the manual override
    /// (Task 2). The result is authoritative about MEMBERSHIP; it is never mutated here.
    public mutating func apply(_ newResult: SequenceResult) { ... }

    /// Synchronous full re-derivation. Three lines: read `sequencerInput`, call the sequencer
    /// with `crate.bpmTolerance`, `apply` the result.
    public mutating func resequence(using sequencer: some Sequencing) {
        apply(sequencer.sequence(candidates: sequencerInput, bpmTolerance: crate.bpmTolerance))
    }
}
```

`apply` in Task 1 is just `result = newResult; rows = CrateRows.make(result: newResult, crate: crate)`.
Task 2 inserts the manual-order reconciliation between those two lines. Do not pre-build the
reconciliation here — let Task 2's RED test drive it.

Note `sequencerInput` deliberately does **not** reuse `crate.allCandidates`: that property offers
every candidate unconditionally and would defeat the pin. Leave `allCandidates` alone; other
call sites use it.

Write this comment above the reconciliation, verbatim — it is the one piece of this phase a
future reader cannot infer from the code:

> **Membership is the sequencer's; arrangement is the user's.**
> `apply(_:)` re-derives which tracks are IN the set — always the sequencer's call, so a
> previously excluded record still rejoins the moment a new addition bridges the gap.
> `manualOrder` re-derives only the ORDER of the ones that are in: filter it down to IDs still
> present in `result.sequence`, then append any newly-sequenced IDs, in the sequencer's own
> order, at the end.

Implementation shape:

```swift
private mutating func rebuildRows() {
    let base = CrateRows.make(result: result, crate: crate)
    guard let manual = crate.manualOrder else { rows = base; return }

    let survivors = manual.filter { id in base.sequenced.contains { $0.id == id } }
    let newcomers = base.sequenced.map(\.id).filter { !survivors.contains($0) }
    let order = survivors + newcomers
    crate.manualOrder = order                       // written in exactly ONE place

    let byID = Dictionary(uniqueKeysWithValues: base.sequenced.map { ($0.id, $0) })
    let reordered = order.compactMap { byID[$0] }.enumerated().map { index, row in
        SequencedRow(id: row.id, vinylID: row.vinylID, indexLabel: String(index + 1),
                     title: row.title, badges: row.badges)
    }
    rows = CrateRows(sequenced: reordered, notIncluded: base.notIncluded)
}
```

`survivors.contains` on an array is O(n²) over the set size. The set is under 100 rows by
construction (MVP-SPEC §5). Leave a `// ponytail:` comment naming that ceiling and pointing at a
`Set` lookup as the upgrade, and move on.

`apply(_:)` becomes `result = newResult; rebuildRows()`. `moveRows` computes the new order from
the *currently displayed* `rows.sequenced` (so a drag on an already-manual list composes),
assigns it to `crate.manualOrder`, then calls `rebuildRows()`. Do not reorder `crate.vinyls`.

Add a second `// ponytail:` comment naming the newcomer rule's ceiling: appending at the end is
the predictable rule; inserting newcomers at their computed slot inside a manual order is
arguably nicer and strictly harder to reason about. Revisit only if the owner asks.
